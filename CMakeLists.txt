cmake_minimum_required(VERSION 3.24)
project(QingYuePDF VERSION 1.0.0 LANGUAGES CXX RC)
add_executable(QingYuePDF WIN32 src/main.cpp src/app.rc)
target_compile_features(QingYuePDF PRIVATE cxx_std_17)
target_compile_definitions(QingYuePDF PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
option(QINGYUE_OLE_TRACE "Temporary local OLE diagnostic log" OFF)
if(QINGYUE_OLE_TRACE)
target_compile_definitions(QingYuePDF PRIVATE QINGYUE_OLE_TRACE)
endif()
target_compile_options(QingYuePDF PRIVATE /W4 /utf-8 /permissive- /EHsc)
set_property(TARGET QingYuePDF PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(QingYuePDF PRIVATE windowsapp comctl32 comdlg32 shell32 shlwapi gdi32 user32 ole32 advapi32 uuid imm32)
target_link_options(QingYuePDF PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /MANIFEST:NO)
set_target_properties(QingYuePDF PROPERTIES OUTPUT_NAME "包子PDF" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist")
add_executable(OleProbe tests/ole_probe.cpp)
target_compile_features(OleProbe PRIVATE cxx_std_17)
target_compile_definitions(OleProbe PRIVATE UNICODE _UNICODE)
target_compile_options(OleProbe PRIVATE /utf-8)
target_link_libraries(OleProbe PRIVATE ole32 uuid)
