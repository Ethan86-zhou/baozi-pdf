cmake_minimum_required(VERSION 3.24)
project(BaoziPDF VERSION 1.1.0 LANGUAGES CXX RC)
set(PDFium_DIR "${CMAKE_SOURCE_DIR}/third_party/pdfium")
find_package(PDFium CONFIG REQUIRED)
add_library(TextSearch STATIC src/text_search.cpp)
target_compile_features(TextSearch PUBLIC cxx_std_17)
target_compile_definitions(TextSearch PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(TextSearch PRIVATE /W4 /utf-8 /permissive- /EHsc)
set_property(TARGET TextSearch PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(TextSearch PUBLIC pdfium user32)
add_executable(BaoziPDF WIN32 src/main.cpp src/app.rc)
target_compile_features(BaoziPDF PRIVATE cxx_std_17)
target_compile_definitions(BaoziPDF PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(BaoziPDF PRIVATE /W4 /utf-8 /permissive- /EHsc)
set_property(TARGET BaoziPDF PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(BaoziPDF PRIVATE TextSearch windowsapp comctl32 comdlg32 shell32 shlwapi gdi32 user32 ole32 advapi32 uuid imm32 msimg32)
target_link_options(BaoziPDF PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /MANIFEST:NO)
set_target_properties(BaoziPDF PROPERTIES OUTPUT_NAME "包子PDF" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist")
add_custom_command(TARGET BaoziPDF POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFium_DIR}/bin/pdfium.dll" "$<TARGET_FILE_DIR:BaoziPDF>/pdfium.dll"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFium_DIR}/LICENSE" "$<TARGET_FILE_DIR:BaoziPDF>/PDFium-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${PDFium_DIR}/licenses" "$<TARGET_FILE_DIR:BaoziPDF>/licenses")
add_executable(OleProbe tests/ole_probe.cpp)
target_compile_features(OleProbe PRIVATE cxx_std_17)
target_compile_definitions(OleProbe PRIVATE UNICODE _UNICODE)
target_compile_options(OleProbe PRIVATE /utf-8)
target_link_libraries(OleProbe PRIVATE ole32 uuid)
add_executable(NavigationTest tests/navigation_test.cpp)
target_compile_features(NavigationTest PRIVATE cxx_std_17)
enable_testing()
add_test(NAME wheel-navigation COMMAND NavigationTest)
add_executable(SearchTest tests/search_test.cpp)
target_compile_features(SearchTest PRIVATE cxx_std_17)
target_compile_definitions(SearchTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(SearchTest PRIVATE /W4 /utf-8 /permissive- /EHsc)
set_property(TARGET SearchTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(SearchTest PRIVATE TextSearch)
add_custom_command(TARGET SearchTest POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFium_DIR}/bin/pdfium.dll" "$<TARGET_FILE_DIR:SearchTest>/pdfium.dll")
add_test(NAME text-search COMMAND SearchTest "${CMAKE_SOURCE_DIR}/tests/fixtures")
